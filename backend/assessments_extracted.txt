2مقدمة
تواصــل وزارة الرتبيــة والتعليــم والتعليــم الفنــي خطواتهــا الحثيثــة نحــو تطويــر منظومــة التعليــم املــري، واالنتقــال بهــا مــن ضفــاف التلقــن 
والحفــظ اآليل إىل رحــاب الفهــم العميــق، والتفكــر الناقــد، وبنــاء املهــارات الشــاملة. ويف هــذا اإلطــار، يســعدنا أن نضــع بــن أيــدي أبنائنــا 
، الــذي أُعِــدَّ بأســلوب منهجــي متكامــل؛ ليكــون مســاعدًا »األداءات والتقييــات « الطــالب، وزمالئنــا املعلمــن، وأوليــاء األمــور الكــرام، كتــاب 
ورفيقًــا ميدانيًّــا طــوال الفصــل الــدرايس. 
لقــد جــرى تصميــم هــذا الكتــاب ليكــون أداة تطبيقيــة فعّالــة ترتجــم أهــداف املنظومــة التعليميــة الحديثــة إىل مارســات واقعيــة داخــل الصــف 
وخارجــه، وذلــك مــن خــالل ثالثــة محــاور رئيســة تحقــق التكامــل يف العمليــة التعليميــة: 
. األداءات الصفيــة (التقييــم البنــايئ املســتمر):وتتضمن أنشــطة متنوعــة يوظفهــا املعلــم يف أثنــاء رشح املــادة داخــل الصــف الــدرايس؛ لتعزيــز 1
 تفاعــل الطــالب، وإتاحــة فــرص مســتمرة للتقييــم البنــايئ، والوقــوف بصــورة مبــارشة عــى مــدى تقدمهــم، وتحديــد جوانــب القــوة والجوانــب 
التــي تحتــاج إىل تحســن؛ مبــا يســاعد عــى تقديــم التغذيــة الراجعــة والدعــم املناســبن يف الوقــت املالئــم. 
. األداءات املنزليــة (التمكــن والتطبيــق الذايت):وهــي مجموعــة مــن األنشــطة والتدريبــات املوجَّهــة، املصممــة ملســاعدة الطــالب عــى التعلــم 2
 الــذايت، والتحقــق مــن مــدى فهمهــم للمعــارف وإتقانهــم للمهــارات التــي تعلموهــا داخــل الصــف، وتطبيقهــا يف ســياقات جديــدة؛ مبــا يرســخ 
التعلــم ويعمــق أثــره. 
. التقييــات األســبوعية (قيــاس األداء واملتابعة):وهــي منــاذج تقييميــة متنوعــة تُنفَّــذ يف نهايــة كل أســبوع؛ ملســاعدة املعلــم واإلدارة املدرســية 3
 عــى متابعــة تقــدم الطــالب، وقيــاس مــدى تحقــق نواتــج التعلــم املســتهدفة فيــا تــم تعلمــه خــالل األســبوع، بصــورة دوريــة ومنتظمــة، مبــا 
يســهم يف تحديــد االحتياجــات التعليميــة واتخــاذ اإلجــراءات الداعمــة لتحســن التعلــم. 
كــا يهــدف الكتــاب، يف الوقــت ذاتــه، إىل إرشاك ويل األمــر بوصفــه رشيــكًا أصيــالً يف العمليــة التعليميــة؛ إذ ميكِّنــه مــن املتابعــة الدقيقــة والواعيــة 
ملســتوى تقــدم ابنــه الــدرايس أوالً بــأول، والتعــرف إىل جوانــب القــوة والجوانــب التــي تحتــاج إىل دعــم، وتقديــم التوجيــه واملســاندة املناســبن 
يف الوقــت املالئــم، دون االقتصــار عــى نتائــج نهايــة الفصــل الــدرايس. 
وتتســم األنشــطة واألســئلة الــواردة يف هــذا الكتــاب باالرتبــاط املبــارش بنواتــج التعلــم املســتهدفة، والتنــوع املتــوازن يف أمناطهــا ومســتوياتها؛ 
فتتضمــن أســئلة موضوعيــة ومقاليــة ومهــام متنوعــة، صُمِّمــت لقيــاس مســتويات مختلفــة مــن التعلــم، بــدءًا مــن التذكــر والفهــم والتطبيــق، 
وصــوالً إىل مهــارات التفكــر العليــا، مثــل التحليــل، والتفكــر الناقــد، وحــل املشــكالت، وصياغــة الــرأي املدعــوم باألدلــة، واإلبــداع. 
وختامًــا، نســأل اللــه تعــاىل أن يكــون هــذا العمــل عونًــا ألبنائنــا الطــالب، ونــورًا يــيء طريقهــم نحــو التفــوق، وســندًا ملعلمينــا األفاضــل يف أداء 
رســالتهم الســامية.
